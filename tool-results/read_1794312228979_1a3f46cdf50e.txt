    88→function pickMime(): { mime: string; ext: string } {
    89→  const candidates = [
    90→    "video/mp4;codecs=avc1.42E01E,mp4a.40.2",
    91→    "video/mp4",
    92→    "video/webm;codecs=vp9,opus",
    93→    "video/webm;codecs=vp8,opus",
    94→    "video/webm",
    95→  ];
    96→  if (typeof MediaRecorder === "undefined") return { mime: "", ext: "webm" };
    97→  for (const m of candidates) {
    98→    if (MediaRecorder.isTypeSupported(m)) {
    99→      return { mime: m, ext: m.startsWith("video/mp4") ? "mp4" : "webm" };
   100→    }
   101→  }
   102→  return { mime: "", ext: "webm" };
   103→}
