    17→function circle(ctx: CanvasRenderingContext2D, x: number, y: number, r: number, color: string) {
    18→  ctx.fillStyle = color;
    19→  ctx.beginPath();
    20→  ctx.arc(x, y, r, 0, Math.PI * 2);
    21→  ctx.fill();
    22→}
    23→
    24→function rr(
    25→  ctx: CanvasRenderingContext2D,
    26→  x: number,
    27→  y: number,
    28→  w: number,
    29→  h: number,
    30→  r: number,
    31→) {
    32→  ctx.beginPath();
    33→  ctx.moveTo(x + r, y);
    34→  ctx.arcTo(x + w, y, x + w, y + h, r);
    35→  ctx.arcTo(x + w, y + h, x, y + h, r);
    36→  ctx.arcTo(x, y + h, x, y, r);
    37→  ctx.arcTo(x, y, x + w, y, r);
    38→  ctx.closePath();
    39→  ctx.fill();
    40→}

    42→function skyGradient(
    43→  ctx: CanvasRenderingContext2D,
    44→  top: string,
    45→  bottom: string,
    46→  horizon: number,
    47→) {
    48→  const g = ctx.createLinearGradient(0, 0, 0, horizon);
    49→  g.addColorStop(0, top);
    50→  g.addColorStop(1, bottom);
    51→  ctx.fillStyle = g;
    52→  ctx.fillRect(0, 0, STAGE_W, horizon);
    53→}

    75→function palm(
    76→  ctx: CanvasRenderingContext2D,
    77→  x: number,
    78→  baseY: number,
    79→  h: number,
    80→  trunk = '#8b5e34',
    81→  leaf = '#15803d',
    82→) {
    83→  ctx.strokeStyle = trunk;
    84→  ctx.lineWidth = 10;
    85→  ctx.lineCap = 'round';
    86→  ctx.beginPath();
    87→  ctx.moveTo(x, baseY);
    88→  ctx.quadraticCurveTo(x + 12, baseY - h * 0.6, x + 6, baseY - h);
    89→  ctx.stroke();
    90→  ctx.fillStyle = leaf;
    91→  for (let i = 0; i < 5; i++) {
    92→    const a = -Math.PI / 2 + (i - 2) * 0.5;
    93→    const lx = x + 6 + Math.cos(a) * 58;
    94→    const ly = baseY - h + Math.sin(a) * 44;
    95→    ctx.beginPath();
    96→    ctx.ellipse(lx, ly, 34, 9, a + Math.PI / 2, 0, Math.PI * 2);
    97→    ctx.fill();
    98→  }
    99→}

   881→// ---------------- ambient life (dynamic overlay, drawn per frame) ----------------
   882→
   883→/**
   884→ * Draws the living layer of a background: drifting clouds, twinkling stars,
   885→ * fireflies, water shimmer, dust motes. Called every frame with scene time.
   886→ */
   887→export function drawAmbient(
   888→  ctx: CanvasRenderingContext2D,
   889→  name: BgName,
   890→  t: number,
   891→) {
   892→  const drift = (speed: number, y: number, s: number, alpha: number, offset: number) => {
   893→    const x = ((t * speed + offset) % (STAGE_W + 360)) - 180;
   894→    ctx.globalAlpha = alpha;
   895→    cloud(ctx, x, y, s);
   896→    ctx.globalAlpha = 1;
   897→  };
   898→
   899→  switch (name) {
   900→    case 'village':
   901→      drift(14, 110, 0.9, 0.9, 0);
   902→      drift(9, 180, 0.6, 0.75, 600);
   903→      // birds
   904→      ctx.strokeStyle = 'rgba(60, 50, 40, 0.7)';
   905→      ctx.lineWidth = 2.4;
   906→      ctx.lineCap = 'round';
   907→      for (let i = 0; i < 3; i++) {
   908→        const bx = ((t * 26 + i * 340) % (STAGE_W + 200)) - 100;
   909→        const by = 130 + Math.sin(t * 2 + i) * 14 + i * 36;
   910→        const flap = Math.sin(t * 7 + i * 2) * 5;
   911→        ctx.beginPath();
   912→        ctx.moveTo(bx - 9, by - flap);
   913→        ctx.quadraticCurveTo(bx, by + 3, bx + 9, by - flap);
   914→        ctx.stroke();
   915→      }
   916→      break;
   917→    case 'field':
   918→      drift(11, 100, 0.8, 0.85, 200);
   919→      drift(16, 170, 0.55, 0.7, 900);
   920→      break;
   921→    case 'bazaar':
   922→      drift(7, 90, 0.7, 0.6, 300);
   923→      break;
   924→    case 'pond':
   925→    case 'ghat': {
   926→      // water shimmer
   927→      ctx.strokeStyle = 'rgba(255, 255, 255, 0.55)';
   928→      ctx.lineWidth = 3;
   929→      ctx.lineCap = 'round';
   930→      for (let i = 0; i < 7; i++) {
   931→        const y = 470 + i * 16;
   932→        const x = 160 + ((i * 260 + t * 40) % 1000);
   933→        ctx.globalAlpha = 0.3 + 0.3 * Math.sin(t * 3 + i);
   934→        ctx.beginPath();
   935→        ctx.moveTo(x, y);
   936→        ctx.quadraticCurveTo(x + 36, y - 5, x + 76, y);
   937→        ctx.stroke();
   938→      }
   939→      ctx.globalAlpha = 1;
   940→      if (name === 'ghat') drift(6, 90, 0.6, 0.5, 500);
   941→      break;
   942→    }
   943→    case 'night': {
   944→      // twinkling stars
   945→      for (let i = 0; i < 12; i++) {
   946→        const x = (i * 397 + 60) % STAGE_W;
   947→        const y = 40 + ((i * 149) % 330);
   948→        ctx.globalAlpha = 0.35 + 0.6 * Math.abs(Math.sin(t * 1.8 + i * 1.7));
   949→        circle(ctx, x, y, 2, '#fef9c3');
   950→      }
   951→      ctx.globalAlpha = 1;
   952→      // fireflies
   953→      for (let i = 0; i < 8; i++) {
   954→        const fx = 200 + ((i * 351 + t * 22) % 900);
   955→        const fy = 480 + Math.sin(t * 1.4 + i * 2.1) * 60 + (i % 3) * 40;
   956→        const glow = 0.4 + 0.6 * Math.abs(Math.sin(t * 2.6 + i));
   957→        ctx.globalAlpha = glow * 0.75;
   958→        ctx.fillStyle = '#bef264';
   959→        ctx.beginPath();
   960→        ctx.arc(fx, fy, 3.2, 0, Math.PI * 2);
   961→        ctx.fill();
   962→        ctx.globalAlpha = glow * 0.2;
   963→        ctx.beginPath();
   964→        ctx.arc(fx, fy, 9, 0, Math.PI * 2);
   965→        ctx.fill();
   966→      }
   967→      ctx.globalAlpha = 1;
   968→      break;
   969→    }
   970→    case 'room':
   971→    case 'school':
   972→    case 'rajcourt': {
   973→      // floating dust motes in warm light
   974→      for (let i = 0; i < 10; i++) {
   975→        const x = 200 + ((i * 263 + t * 9) % 880);
   976→        const y = 180 + Math.sin(t * 0.9 + i * 1.3) * 120 + ((i * 53) % 160);
   977→        ctx.globalAlpha = 0.12 + 0.1 * Math.sin(t * 1.6 + i);
   978→        circle(ctx, x, y, 2.4, '#fff7ed');
   979→      }
   980→      ctx.globalAlpha = 1;
   981→      break;
   982→    }
   983→  }
   984→}

  1000→/** Get (or build) the cached background canvas. Client-side only. */
  1001→export function getBackgroundCanvas(name: BgName): HTMLCanvasElement {
  1002→  if (typeof document === 'undefined') throw new Error('backgrounds require a browser');
  1003→  let c = cache.get(name);
  1004→  if (!c) {
  1005→    c = baseCanvas();
  1006→    const ctx = c.getContext('2d')!;
  1007→    RENDERERS[name](ctx);
  1008→    cache.set(name, c);
  1009→  }
  1010→  return c;
  1011→}
