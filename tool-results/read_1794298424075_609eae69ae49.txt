    56→function pickMime(): { mime: string; ext: string } {
    57→  const candidates = [
    58→    "video/mp4;codecs=avc1.42E01E,mp4a.40.2",
    59→    "video/mp4",
    60→    "video/webm;codecs=vp9,opus",
    61→    "video/webm;codecs=vp8,opus",
    62→    "video/webm",
    63→  ];
    64→  if (typeof MediaRecorder === "undefined") return { mime: "", ext: "webm" };
    65→  for (const m of candidates) {
    66→    if (MediaRecorder.isTypeSupported(m)) {
    67→      return { mime: m, ext: m.startsWith("video/mp4") ? "mp4" : "webm" };
    68→    }
    69→  }
    70→  return { mime: "", ext: "webm" };
    71→}
