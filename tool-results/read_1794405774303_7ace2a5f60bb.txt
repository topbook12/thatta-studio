    38→/** sequential start fallback for beats without an absolute `start` */
    39→function seqStart(beats: SceneBeat[], i: number): number {
    40→  let acc = 0;
    41→  for (let j = 0; j < i; j++) acc += Math.max(0.3, beats[j].duration);
    42→  return acc;
    43→}

    45→function wrapText(
    46→  ctx: CanvasRenderingContext2D,
    47→  text: string,
    48→  maxWidth: number,
    49→): string[] {
    50→  const words = text.split(/\s+/).filter(Boolean);
    51→  const lines: string[] = [];
    52→  let cur = '';
    53→  for (const w of words) {
    54→    const test = cur ? cur + ' ' + w : w;
    55→    if (ctx.measureText(test).width > maxWidth && cur) {
    56→      lines.push(cur);
    57→      cur = w;
    58→    } else {
    59→      cur = test;
    60→    }
    61→  }
    62→  if (cur) lines.push(cur);
    63→  return lines;
    64→}

   131→  setScene(scene: Scene, designs: CharacterDesign[]) {
   132→    this.scene = scene;
   133→    this.designs = new Map(designs.map((d) => [d.id, d]));
   134→    // keep the editable walk-flag if its beat still exists and is a walk/run
   135→    if (this.editBeatId) {
   136→      const b = scene.beats.find((x) => x.id === this.editBeatId);
   137→      if (!b || (b.action !== 'walk' && b.action !== 'run')) this.editBeatId = null;
   138→    }
   139→    if (this.scene.beats.length === 0) {
   140→      this.sceneTime = 0;
   141→      this.activeIds.clear();
   142→      this.callbacks.onTime?.(0, 0);
   143→      return;
   144→    }
   145→    // preserve the playhead position across scene edits (never rewind!)
   146→    const total = this.totalDuration();
   147→    const clamped = Math.min(this.sceneTime, Math.max(0, total - 0.001));
   148→    if (Math.abs(clamped - this.sceneTime) > 0.001) {
   149→      this.seek(clamped);
   150→    } else {
   151→      // re-sync active windows for the new layout (no audio restart)
   152→      const active = this.activeAt(this.layout(), this.sceneTime);
   153→      this.activeIds = new Set(active.map((w) => w.beat.id));
   154→      this.callbacks.onTime?.(this.sceneTime, active.length);
   155→    }
   156→  }

   203→  // ---------------- timeline layout ----------------
   204→
   205→  /** All beat windows sorted by start time (parallel-safe). */
   206→  private layout(): BeatWindow[] {
   207→    if (!this.scene) return [];
   208→    const beats = this.scene.beats;
   209→    const wins = beats.map((b, i) => {
   210→      const start =
   211→        typeof b.start === 'number' && b.start >= 0 ? b.start : seqStart(beats, i);
   212→      const dur = Math.max(0.3, this.durations[b.id] ?? b.duration);
   213→      return { beat: b, idx: i, start, end: start + dur } as BeatWindow;
   214→    });
   215→    wins.sort((a, b) => a.start - b.start || a.idx - b.idx);
   216→    return wins;
   217→  }
   218→
   219→  private totalDuration(): number {
   220→    let total = 0;
   221→    for (const w of this.layout()) total = Math.max(total, w.end);
   222→    return total;
   223→  }
   224→
   225→  private activeAt(wins: BeatWindow[], t: number): BeatWindow[] {
   226→    return wins.filter((w) => w.start <= t && t < w.end);
   227→  }
   228→
   229→  /**
   230→   * Where does this character stand at scene time `t`?
   231→   * Deterministic walk-chain: x = the destination of the latest walk that
   232→   * already finished before t (so chained walks never teleport).
   233→   */
   234→  private charXAt(uid: string, t: number, placed: PlacedCharacter): number {
   235→    let x = (placed.x / 100) * STAGE_W;
   236→    let best: BeatWindow | null = null;
   237→    for (const w of this.layout()) {
   238→      if (
   239→        w.beat.actorUid === uid &&
   240→        (w.beat.action === 'walk' || w.beat.action === 'run') &&
   241→        w.beat.moveTo != null &&
   242→        w.end <= t + 1e-6
   243→      ) {
   244→        if (!best || w.end > best.end) best = w;
   245→      }
   246→    }
   247→    if (best && best.beat.moveTo != null) x = (best.beat.moveTo / 100) * STAGE_W;
   248→    return x;
   249→  }
   250→
   251→  // ---------------- transport ----------------
   252→
   253→  /** Jump to an absolute scene time (works while playing or paused). */
   254→  seek(t: number) {
   255→    const wins = this.layout();
   256→    if (wins.length === 0) return;
   257→    const total = this.totalDuration();
   258→    const clamped = clamp(t, 0, Math.max(0, total - 0.001));
   259→    this.sceneTime = clamped;
   260→    const active = this.activeAt(wins, clamped);
   261→    const newIds = new Set(active.map((w) => w.beat.id));
   262→
   263→    if (this.playing) {
   264→      for (const id of this.activeIds) {
   265→        if (!newIds.has(id)) {
   266→          const w = wins.find((x) => x.beat.id === id);
   267→          if (w) this.callbacks.onBeatEnd?.(w.beat);
   268→        }
   269→      }
   270→      for (const w of active) {
   271→        if (!this.activeIds.has(w.beat.id)) {
   272→          this.callbacks.onBeatStart?.(w.beat, Math.max(0, clamped - w.start));
   273→        }
   274→      }
   275→    }
   276→    this.activeIds = newIds;
   277→    this.callbacks.onTime?.(clamped, active.length);
   278→  }
   279→
   280→  play(): boolean {
   281→    const wins = this.layout();
   282→    if (wins.length === 0) return false;
   283→    if (this.sceneTime >= this.totalDuration() - 0.02) this.sceneTime = 0;
   284→    const active = this.activeAt(wins, this.sceneTime);
   285→    this.playing = true;
   286→    this.activeIds = new Set(active.map((w) => w.beat.id));
   287→    for (const w of active) {
   288→      this.callbacks.onBeatStart?.(w.beat, Math.max(0, this.sceneTime - w.start));
   289→    }
   290→    this.callbacks.onTime?.(this.sceneTime, active.length);
   291→    return true;
   292→  }
   293→
   294→  pause() {
   295→    if (!this.playing) return;
   296→    this.playing = false;
   297→    const wins = this.layout();
   298→    for (const id of this.activeIds) {
   299→      const w = wins.find((x) => x.beat.id === id);
   300→      if (w) this.callbacks.onBeatEnd?.(w.beat);
   301→    }
   302→    this.activeIds = new Set();
   303→  }
   304→
   305→  stop() {
   306→    this.pause();
   307→    this.sceneTime = 0;
   308→    this.callbacks.onTime?.(0, 0);
   309→  }

   318→  private canvasY(e: PointerEvent): number {

   323→  /** x of the editable walk-target flag, or null when not shown */
   324→  private flagX(): number | null {
   325→    if (this.playing || !this.editBeatId || !this.scene) return null;
   326→    const beat = this.scene.beats.find((b) => b.id === this.editBeatId);
   327→    if (!beat) return null;
   328→    if (beat.action !== 'walk' && beat.action !== 'run') return null;
   329→    if (beat.moveTo == null) return null;
   330→    return (beat.moveTo / 100) * STAGE_W;
   331→  }
   332→
   333→  private overFlag(x: number, y: number): boolean {
   334→    const fx = this.flagX();
   335→    if (fx == null) return false;
   336→    return Math.abs(x - fx) <= 30 && y >= GROUND_Y - 92 && y <= GROUND_Y + 18;
   337→  }
   338→
   339→  private hitCharacter(x: number, y: number): PlacedCharacter | null {
   340→    if (!this.scene) return null;
   341→    for (let i = this.scene.characters.length - 1; i >= 0; i--) {
   342→      const c = this.scene.characters[i];
   343→      const cx = (c.x / 100) * STAGE_W;
   344→      const s = c.scale;
   345→      if (Math.abs(x - cx) < 60 * s && y > GROUND_Y - 245 * s && y < GROUND_Y + 14) {
   346→        return c;
   347→      }
   348→    }
   349→    return null;
   350→  }

   352→  private attachPointer() {
   353→    this.canvas.addEventListener('pointerdown', this.onPointerDown);
   354→    this.canvas.addEventListener('pointermove', this.onPointerMove);
   355→    this.canvas.addEventListener('pointerup', this.onPointerUp);
   356→    this.canvas.addEventListener('pointercancel', this.onPointerUp);
   357→  }
   358→
   359→  private detachPointer() {
   360→    this.canvas.removeEventListener('pointerdown', this.onPointerDown);
   361→    this.canvas.removeEventListener('pointermove', this.onPointerMove);
   362→    this.canvas.removeEventListener('pointerup', this.onPointerUp);
   363→    this.canvas.removeEventListener('pointercancel', this.onPointerUp);
   364→  }
   365→
   366→  private onPointerDown = (e: PointerEvent) => {
   367→    const x = this.canvasX(e);
   368→    const y = this.canvasY(e);
   369→    // walk-target flag has priority
   370→    if (this.overFlag(x, y)) {
   371→      this.flagDrag = true;
   372→      this.canvas.setPointerCapture(e.pointerId);
   373→      this.interactions.onDragTarget?.(clamp((x / STAGE_W) * 100, 4, 96));
   374→      this.canvas.style.cursor = 'ew-resize';
   375→      return;
   376→    }
   377→    const hit = this.hitCharacter(x, y);
   378→    if (hit) {
   379→      this.dragUid = hit.uid;
   380→      this.dragMoved = false;
   381→      this.selectedUid = hit.uid;
   382→      this.canvas.setPointerCapture(e.pointerId);
   383→      this.interactions.onPick?.(hit.uid);
   384→    } else {
   385→      this.selectedUid = null;
   386→      this.interactions.onPick?.(null);
   387→    }
   388→  };
   389→
   390→  private onPointerMove = (e: PointerEvent) => {
   391→    const x = this.canvasX(e);
   392→    const y = this.canvasY(e);
   393→    if (this.flagDrag) {
   394→      this.interactions.onDragTarget?.(clamp((x / STAGE_W) * 100, 4, 96));
   395→      this.canvas.style.cursor = 'ew-resize';
   396→      return;
   397→    }
   398→    if (this.dragUid) {
   399→      this.dragMoved = true;
   400→      const pct = clamp((x / STAGE_W) * 100, 4, 96);
   401→      this.interactions.onDrag?.(this.dragUid, pct);
   402→      this.canvas.style.cursor = 'grabbing';
   403→      return;
   404→    }
   405→    if (this.overFlag(x, y)) {
   406→      this.canvas.style.cursor = 'ew-resize';
   407→      return;
   408→    }
   409→    const hit = this.hitCharacter(x, y);
   410→    this.canvas.style.cursor = hit ? 'grab' : 'default';
   411→  };
   412→
   413→  private onPointerUp = (e: PointerEvent) => {
   414→    if (this.flagDrag) {
   415→      try {
   416→        this.canvas.releasePointerCapture(e.pointerId);
   417→      } catch {
   418→        /* ignore */
   419→      }
   420→      this.flagDrag = false;
   421→      const pct = clamp((this.canvasX(e) / STAGE_W) * 100, 4, 96);
   422→      this.interactions.onDragTargetEnd?.(pct);
   423→      this.canvas.style.cursor = 'default';
   424→      return;
   425→    }
   426→    if (this.dragUid) {
   427→      try {
   428→        this.canvas.releasePointerCapture(e.pointerId);
   429→      } catch {
   430→        /* ignore */
   431→      }
   432→      const pct = clamp((this.canvasX(e) / STAGE_W) * 100, 4, 96);
   433→      if (this.dragMoved) this.interactions.onDragEnd?.(this.dragUid, pct);
   434→      this.dragUid = null;
   435→      this.canvas.style.cursor = 'grab';
   436→    }
   437→  };
   438→
   439→  // ---------------- main loop ----------------
   440→
   441→  private loop = (ts: number) => {
   442→    this.raf = requestAnimationFrame(this.loop);
   443→    const dt = Math.min(0.1, (ts - this.lastTs) / 1000);
   444→    this.lastTs = ts;
   445→    this.animT += dt * this.rate;
   446→
   447→    if (this.playing && this.scene && this.scene.beats.length > 0) {
   448→      this.sceneTime += dt * this.rate;
   449→      const wins = this.layout();
   450→      const total = this.totalDuration();
   451→      if (this.sceneTime >= total) {
   452→        this.sceneTime = total;
   453→        const stillActive = wins.filter((w) => this.activeIds.has(w.beat.id));
   454→        for (const w of stillActive) this.callbacks.onBeatEnd?.(w.beat);
   455→        this.activeIds.clear();
   456→        this.playing = false;
   457→        this.callbacks.onTime?.(this.sceneTime, 0);
   458→        this.callbacks.onFinish?.();
   459→      } else {
   460→        const active = this.activeAt(wins, this.sceneTime);
   461→        const newIds = new Set(active.map((w) => w.beat.id));
   462→        for (const id of this.activeIds) {
   463→          if (!newIds.has(id)) {
   464→            const w = wins.find((x) => x.beat.id === id);
   465→            if (w) this.callbacks.onBeatEnd?.(w.beat);
   466→          }
   467→        }
   468→        for (const w of active) {
   469→          if (!this.activeIds.has(w.beat.id)) {
   470→            this.callbacks.onBeatStart?.(w.beat, Math.max(0, this.sceneTime - w.start));
   471→          }
   472→        }
   473→        this.activeIds = newIds;
   474→        this.callbacks.onTime?.(this.sceneTime, active.length);
   475→      }
   476→    }
   477→
   478→    this.render();
   479→  };

   481→  // ---------------- rendering ----------------
   482→
   483→  private render() {
   484→    const ctx = this.ctx;
   485→    ctx.clearRect(0, 0, STAGE_W, STAGE_H);
   486→
   487→    if (!this.scene || this.scene.beats.length === 0) {
   488→      // empty stage splash
   489→      ctx.fillStyle = '#fff7ed';
   490→      ctx.fillRect(0, 0, STAGE_W, STAGE_H);
   491→      ctx.fillStyle = '#9a3412';
   492→      ctx.font = '700 42px "Hind Siliguri", sans-serif';
   493→      ctx.textAlign = 'center';
   494→      ctx.textBaseline = 'middle';
   495→      ctx.fillText('ঠাট্টা স্টুডিও', STAGE_W / 2, STAGE_H / 2 - 20);
   496→      ctx.font = '500 26px "Hind Siliguri", sans-serif';
   497→      ctx.fillStyle = '#c2410c';
   498→      ctx.fillText(
   499→        'ক্যারেক্টার বসিয়ে কার্টুন বানানো শুরু করুন',
   500→        STAGE_W / 2,
   501→        STAGE_H / 2 + 28,
   502→      );
   503→      return;
   504→    }
   505→
   506→    // background
   507→    try {
   508→      ctx.drawImage(getBackgroundCanvas(this.scene.background), 0, 0);
   509→    } catch {
   510→      ctx.fillStyle = '#fde68a';
   511→      ctx.fillRect(0, 0, STAGE_W, STAGE_H);
   512→    }
   513→
   514→    // living background layer (clouds, fireflies, shimmer…)
   515→    try {
   516→      drawAmbient(ctx, this.scene.background, this.animT);
   517→    } catch {
   518→      /* ambient is decorative — never break the stage */
   519→    }
   520→
   521→    const t = this.sceneTime;
   522→    const inScene = this.playing || t > 0;
   523→    const wins = this.layout();
   524→    const active = inScene ? this.activeAt(wins, t) : [];
   525→
   526→    // one active window per actor (latest start wins)
   527→    const activeByActor = new Map<string, BeatWindow>();
   528→    for (const w of active) {
   529→      const prev = activeByActor.get(w.beat.actorUid);
   530→      if (!prev || w.start > prev.start) activeByActor.set(w.beat.actorUid, w);
   531→    }
   532→
   533→    for (const placed of this.scene.characters) {
   534→      const design = this.designs.get(placed.designId);
   535→      if (!design) continue;
   536→
   537→      const w = inScene ? (activeByActor.get(placed.uid) ?? null) : null;
   538→      let worldX = this.charXAt(placed.uid, inScene ? t : 0, placed);
   539→      let flip = placed.flip;
   540→      let pose = basePose();
   541→      let localT = this.animT;
   542→
   543→      if (w) {
   544→        const beat = w.beat;
   545→        const beatElapsed = Math.max(0, t - w.start);
   546→        const speed = clamp(beat.speed ?? 1, 0.25, 2.5);
   547→        localT = beatElapsed * speed;
   548→        pose = computePose(beat.action, localT, beat.text.trim().length > 0);
   549→
   550→        // ---- real viseme lip-sync: audio envelopes drive the mouth ----
   551→        const env = this.envelopes[beat.id];
   552→        const wideEnv = this.wideEnvelopes[beat.id];
   553→        if (env && env.length > 0 && beat.text.trim().length > 0) {
   554→          const fi = beatElapsed * 24;
   555→          const i0 = Math.max(0, Math.min(env.length - 1, Math.floor(fi)));
   556→          const i1 = Math.max(0, Math.min(env.length - 1, i0 + 1));
   557→          const frac = fi - Math.floor(fi);
   558→          const amp = lerp(env[i0], env[i1], frac);
   559→          pose.mouthOpen = Math.max(0.06, Math.min(1, 0.12 + amp * 1.05));
   560→          // “ee/ai” viseme widens the mouth horizontally when present
   561→          if (wideEnv && wideEnv.length > 0) {
   562→            const w0 = wideEnv[Math.max(0, Math.min(wideEnv.length - 1, Math.floor(fi)))];
   563→            const w1 =
   564→              wideEnv[Math.max(0, Math.min(wideEnv.length - 1, Math.floor(fi) + 1))] ?? w0;
   565→            pose.mouthWide = 0.18 + lerp(w0, w1, frac) * 0.82;
   566→          } else {
   567→            pose.mouthWide = 0.25 + amp * 0.35;
   568→          }
   569→          // tiny speech head-bob for liveliness
   570→          pose.headNod += (amp - 0.35) * 2.2;
   571→        }
   572→
   573→        if (
   574→          (beat.action === 'walk' || beat.action === 'run') &&
   575→          beat.moveTo != null
   576→        ) {
   577→          const dur = Math.max(0.3, w.end - w.start);
   578→          const progress = Math.min(1, beatElapsed / dur);
   579→          const eased = easingByName(beat.easing ?? 'easeInOut')(progress);
   580→          const startX = this.charXAt(placed.uid, w.start, placed);
   581→          const targetX = (beat.moveTo / 100) * STAGE_W;
   582→          worldX = lerp(startX, targetX, eased);
   583→          flip = targetX < startX;
   584→        }
   585→      } else {
   586→        pose = computePose('idle', localT, false);
   587→      }
   588→
   589→      const s = placed.scale;
   590→      ctx.save();
   591→      ctx.translate(worldX, GROUND_Y);
   592→      ctx.scale(flip ? -s : s, s);
   593→      drawCharacter(ctx, design, pose, localT, placed.prop ?? 'none');
   594→      ctx.restore();
   595→
   596→      // selection ring
   597→      if (this.selectedUid === placed.uid) {
   598→        ctx.save();
   599→        ctx.strokeStyle = 'rgba(234, 88, 12, 0.85)';
   600→        ctx.lineWidth = 3;
   601→        ctx.setLineDash([9, 7]);
   602→        ctx.beginPath();
   603→        ctx.ellipse(worldX, GROUND_Y + 4, 46 * s, 11 * s, 0, 0, Math.PI * 2);
   604→        ctx.stroke();
   605→        ctx.restore();
   606→      }
   607→
   608→      if (w) {
   609→        this.drawSpeakerPill(design.name, worldX, s);
   610→      }
   611→    }
   612→
   613→    // walk-target flag (only while paused + an editable walk beat selected)
   614→    if (!this.playing && this.editBeatId) {
   615→      const w = wins.find((x) => x.beat.id === this.editBeatId);
   616→      if (
   617→        w &&
   618→        (w.beat.action === 'walk' || w.beat.action === 'run') &&
   619→        w.beat.moveTo != null
   620→      ) {
   621→        const actor = this.scene.characters.find((c) => c.uid === w.beat.actorUid);
   622→        const fromX = actor
   623→          ? this.charXAt(w.beat.actorUid, w.start, actor)
   624→          : STAGE_W / 2;
   625→        this.drawWalkFlag(fromX, w.beat.moveTo);
   626→      }
   627→    }
   628→
   629→    // subtitle: the most recently started active beat with dialogue
   630→    let sub: BeatWindow | null = null;
   631→    for (const w of active) {
   632→      if (w.beat.text.trim() && (!sub || w.start > sub.start)) sub = w;
   633→    }
   634→    if (sub) {
   635→      this.drawSubtitle(sub.beat.text);
   636→    }
   637→
   638→    // watermark
   639→    this.drawWatermark();
   640→  }

   642→  private drawWalkFlag(fromX: number, moveToPct: number) {
   643→    const ctx = this.ctx;
   644→    const flagX = (moveToPct / 100) * STAGE_W;
   645→    ctx.save();
   646→
   647→    // dashed walk path along the ground
   648→    ctx.strokeStyle = 'rgba(5, 150, 105, 0.7)';
   649→    ctx.lineWidth = 3;
   650→    ctx.setLineDash([10, 8]);
   651→    ctx.beginPath();
   652→    ctx.moveTo(fromX, GROUND_Y + 6);
   653→    ctx.lineTo(flagX, GROUND_Y + 6);
   654→    ctx.stroke();
   655→    ctx.setLineDash([]);
   656→
   657→    // pole
   658→    ctx.strokeStyle = '#065f46';
   659→    ctx.lineWidth = 4;
   660→    ctx.beginPath();
   661→    ctx.moveTo(flagX, GROUND_Y + 6);
   662→    ctx.lineTo(flagX, GROUND_Y - 64);
   663→    ctx.stroke();
   664→
   665→    // pennant
   666→    ctx.fillStyle = '#10b981';
   667→    ctx.beginPath();
   668→    ctx.moveTo(flagX, GROUND_Y - 64);
   669→    ctx.lineTo(flagX + 46, GROUND_Y - 52);
   670→    ctx.lineTo(flagX, GROUND_Y - 40);
   671→    ctx.closePath();
   672→    ctx.fill();
   673→
   674→    // draggable base knob
   675→    ctx.fillStyle = '#059669';
   676→    ctx.beginPath();
   677→    ctx.arc(flagX, GROUND_Y + 6, 11, 0, Math.PI * 2);
   678→    ctx.fill();
   679→    ctx.strokeStyle = '#ecfdf5';
   680→    ctx.lineWidth = 3;
   681→    ctx.stroke();
   682→
   683→    // label
   684→    ctx.font = '600 17px "Hind Siliguri", sans-serif';
   685→    ctx.textAlign = 'center';
   686→    ctx.textBaseline = 'middle';
   687→    const labelX = clamp(flagX, 120, STAGE_W - 150);
   688→    ctx.fillStyle = 'rgba(6, 95, 70, 0.92)';
   689→    const label = 'গন্তব্য — টেনে বদলান';
   690→    const w = ctx.measureText(label).width + 20;
   691→    ctx.beginPath();
   692→    ctx.roundRect(labelX - w / 2, GROUND_Y - 92, w, 26, 13);
   693→    ctx.fill();
   694→    ctx.fillStyle = '#ecfdf5';
   695→    ctx.fillText(label, labelX, GROUND_Y - 78);
   696→
   697→    ctx.restore();
   698→  }

   700→  private drawSpeakerPill(name: string, worldX: number, s: number) {
   701→    const ctx = this.ctx;
   702→    const y = GROUND_Y - 232 * s - 16;
   703→    ctx.font = '600 24px "Hind Siliguri", sans-serif';
   704→    ctx.textAlign = 'center';
   705→    ctx.textBaseline = 'middle';
   706→    const w = ctx.measureText(name).width + 34;
   707→    const cx = Math.max(w / 2 + 8, Math.min(STAGE_W - w / 2 - 8, worldX));
   708→    ctx.fillStyle = 'rgba(124, 45, 18, 0.88)';
   709→    ctx.beginPath();
   710→    ctx.roundRect(cx - w / 2, y - 19, w, 38, 19);
   711→    ctx.fill();
   712→    ctx.fillStyle = '#fff7ed';
   713→    ctx.fillText(name, cx, y + 1);
   714→  }

   716→  private drawSubtitle(text: string) {
   717→    const ctx = this.ctx;
   718→    let fontSize = 30;
   719→    ctx.font = `600 ${fontSize}px "Hind Siliguri", sans-serif`;
   720→    const maxW = 980;
   721→    let lines = wrapText(ctx, text, maxW);
   722→    if (lines.length > 2) {
   723→      fontSize = 25;
   724→      ctx.font = `600 ${fontSize}px "Hind Siliguri", sans-serif`;
   725→      lines = wrapText(ctx, text, maxW);
   726→    }
   727→    const lh = fontSize * 1.45;
   728→    const boxH = lines.length * lh + 22;
   729→    const boxW = Math.min(
   730→      maxW + 56,
   731→      Math.max(...lines.map((l) => ctx.measureText(l).width)) + 56,
   732→    );
   733→    const x0 = (STAGE_W - boxW) / 2;
   734→    const y0 = STAGE_H - boxH - 14;
   735→
   736→    ctx.fillStyle = 'rgba(255, 251, 235, 0.95)';
   737→    ctx.strokeStyle = '#f59e0b';
   738→    ctx.lineWidth = 3;
   739→    ctx.beginPath();
   740→    ctx.roundRect(x0, y0, boxW, boxH, 14);
   741→    ctx.fill();
   742→    ctx.stroke();
   743→
   744→    ctx.fillStyle = '#1c1917';
   745→    ctx.textAlign = 'center';
   746→    ctx.textBaseline = 'middle';
   747→    lines.forEach((line, i) => {
   748→      ctx.fillText(line, STAGE_W / 2, y0 + 14 + lh / 2 + i * lh);
   749→    });
   750→  }
   751→
   752→  private drawWatermark() {
   753→    const ctx = this.ctx;
   754→    ctx.font = '600 20px "Hind Siliguri", sans-serif';
   755→    ctx.textAlign = 'left';
   756→    ctx.textBaseline = 'middle';
   757→    const label = '🎬 ঠাট্টা স্টুডিও';
   758→    const w = ctx.measureText(label).width + 26;
   759→    ctx.fillStyle = 'rgba(255, 255, 255, 0.72)';
   760→    ctx.beginPath();
   761→    ctx.roundRect(14, 14, w, 36, 18);
   762→    ctx.fill();
   763→    ctx.fillStyle = '#9a3412';
   764→    ctx.fillText(label, 27, 33);
   765→  }
