     2→// ঠাট্টা স্টুডিও — Pose computation (procedural actions)
     3→// Sprint 2: the 12 classic animation principles pass — follow-through,
     4→// overlapping action, secondary motion, squash & stretch, anticipation.
     5→// Every action is a pure function of t — deterministic for scrubbing/export.
     6→// ============================================================
     7→
     8→import type { ActionName, EasingName, Pose } from './types';
     9→import { applyExpression } from './expression';
    10→
    11→const TAU = Math.PI * 2;
    12→
    13→export function basePose(): Pose {
    14→  return {
    15→    xOff: 0,
    16→    yOff: 0,
    17→    bodyLean: 0,
    18→    headTilt: 0,
    19→    headNod: 0,
    20→    shoulderL: 4,
    21→    elbowL: -6,
    22→    shoulderR: 4,
    23→    elbowR: -6,
    24→    hipL: 0,
    25→    kneeL: 0,
    26→    hipR: 0,
    27→    kneeR: 0,
    28→    mouthOpen: 0,
    29→    mouthWide: 0,
    30→    eyeState: 'open',
    31→    browState: 'normal',
    32→    expression: 'neutral',
    33→    tears: false,
    34→    squash: 0,
    35→  };
    36→}
    37→
    38→// ------------------------------------------------------------
    39→// Animation-principle helpers (used throughout the actions below)
    40→// ------------------------------------------------------------
    41→
    42→/** damped settle oscillation — follow-through after an impact */
    43→const settle = (k: number, freq = 9, decay = 5): number =>
    44→  Math.exp(-decay * k) * Math.cos(freq * k);
    45→
    46→/** phase-lagged sine — overlapping action: a limb trails its parent by `lag` */
    47→const lagSin = (ph: number, lag: number): number => Math.sin(ph - lag);
    48→
    49→/** cubic ease-out — fast start, soft landing (extension phase of `reach`) */
    50→const easeOutCubic = (x: number): number => 1 - Math.pow(1 - x, 3);
    51→
    52→/**
    53→ * ease with slight overshoot then settle (for point/reach gestures).
    54→ * Returns a 0..1 progress multiplier: races to ~1.08 by `hitAt` seconds,
    55→ * then wobbles back onto the target by `total` seconds (follow-through).
    56→ */
    57→function reach(t: number, hitAt: number, total: number): number {
    58→  if (t <= 0) return 0;
    59→  if (t < hitAt) return 1.08 * easeOutCubic(t / hitAt);
    60→  if (t >= total) return 1;
    61→  return 1 + 0.08 * settle((t - hitAt) / (total - hitAt));
    62→}
    63→
    64→/** periodic blink helper (once every ~3.4s) */
    65→function blinking(t: number): boolean {
    66→  return t % 3.4 < 0.12;
    67→}
    68→
    69→/**
    70→ * Compute the pose for an action at time `t` (seconds since action start).
    71→ * `speaking` makes the mouth flap (dialogue playing).
    72→ */
    73→export function computePose(action: ActionName, t: number, speaking: boolean): Pose {
    74→  const p = basePose();
    75→
    76→  switch (action) {
    77→    case 'idle': {
    78→      // breathing (subtle squash & stretch) + slow weight shift
    79→      p.squash = Math.sin(t * 1.7) * 0.035;
    80→      p.xOff = Math.sin(t * 0.45) * 1.4;
    81→      p.yOff = Math.sin(t * 1.8) * 2;
    82→      // shoulders breathe with a micro phase-lag behind the torso (overlapping)
    83→      p.shoulderL = 4 + Math.sin(t * 1.8 - 0.4) * 3;
    84→      p.shoulderR = 4 + Math.sin(t * 1.8 + 0.6) * 3;
    85→      // head drifts on its own slower rhythm (secondary action)
    86→      p.headTilt = Math.sin(t * 0.9 - 0.3) * 2;
    87→      if (blinking(t)) p.eyeState = 'blink';
    88→      if (speaking) {
    89→        p.mouthOpen = 0.15 + 0.75 * Math.abs(Math.sin(t * 12.5));
    90→        p.shoulderR = 8 + Math.sin(t * 3.2) * 10;
    91→        p.elbowR = -18 + Math.sin(t * 3.2 - 0.55) * 12; // forearm lags upper arm
    92→      }
    93→      p.expression = 'neutral';
    94→      break;
    95→    }
    96→    case 'talk': {
    97→      p.yOff = Math.sin(t * 2.2) * 1.8;
    98→      p.xOff = Math.sin(t * 0.5) * 1.2; // weight shift while chatting
    99→      p.mouthOpen = speaking ? 0.15 + 0.8 * Math.abs(Math.sin(t * 13)) : 0.08;
   100→      // gesture arms: forearms trail the upper arms (overlapping action)
   101→      p.shoulderL = 6 + Math.sin(t * 2.8) * 8;
   102→      p.elbowL = -22 + Math.sin(t * 2.8 - 0.5) * 14;
   103→      p.shoulderR = 8 + Math.sin(t * 3.4) * 12;
   104→      p.elbowR = -26 + Math.sin(t * 3.4 - 0.55) * 16;
   105→      // head nod lags the speech rhythm slightly
   106→      p.headNod = Math.sin(t * 2.6 - 0.7) * 4;
   107→      if (blinking(t)) p.eyeState = 'blink';
   108→      p.expression = 'neutral';
   109→      break;
   110→    }
   111→    case 'walk': {
   112→      const ph = t * TAU * 1.8;
   113→      p.hipL = Math.sin(ph) * 26;
   114→      p.hipR = Math.sin(ph + Math.PI) * 26;
   115→      p.kneeL = Math.max(0, Math.sin(ph + 1.2)) * 30;
   116→      p.kneeR = Math.max(0, Math.sin(ph + Math.PI + 1.2)) * 30;
   117→      // arm swing with follow-through: elbows lag the shoulders by ~0.55 rad
   118→      p.shoulderL = 5 + Math.sin(ph + Math.PI) * 24;
   119→      p.shoulderR = 5 + Math.sin(ph) * 24;
   120→      p.elbowL = -12 + lagSin(ph + Math.PI, 0.55) * 14;
   121→      p.elbowR = -12 + lagSin(ph, 0.55) * 14;
   122→      p.yOff = Math.abs(Math.sin(ph)) * 3.5; // heel-lift bounce
   123→      p.xOff = Math.sin(ph - 0.5) * 1.8; // hip sway trails the feet
   124→      p.bodyLean = 3 + Math.sin(ph) * 1.6; // torso roll
   125→      p.headNod = Math.sin(ph * 2 - 0.55) * 1.8; // counter-bob, delayed
   126→      p.squash = Math.max(0, Math.sin(ph * 2)) * 0.05; // footstrike squash
   127→      p.mouthOpen = speaking ? 0.15 + 0.7 * Math.abs(Math.sin(t * 12)) : 0;
   128→      if (blinking(t)) p.eyeState = 'blink';
   129→      break;
   130→    }
   131→    case 'wave': {
   132→      // forearm whip: elbow trails the shoulder oscillation by ~0.55 rad
   133→      p.shoulderR = 152 + Math.sin(t * 9.5) * 10;
   134→      p.elbowR = -14 + Math.sin(t * 9.5 - 0.55) * 26;
   135→      // body counter-lean absorbs the whip (follow-through)
   136→      p.bodyLean = -Math.sin(t * 9.5) * 2.2;
   137→      p.shoulderL = 5 + Math.sin(t * 1.8) * 2; // other arm relaxed sway
   138→      p.elbowL = -8;
   139→      p.headTilt = Math.sin(t * 4.5) * 5;
   140→      p.yOff = Math.sin(t * 2) * 1.5;
   141→      applyExpression(p, 'smile', 0.75);
   142→      p.mouthOpen = speaking ? 0.2 + 0.7 * Math.abs(Math.sin(t * 12)) : 0;
   143→      if (blinking(t)) p.eyeState = 'blink';
   144→      break;
   145→    }
   146→    case 'laugh': {
   147→      // layered bounce: body leads, shoulders and head follow with delay
   148→      p.yOff = Math.abs(Math.sin(t * 8)) * 7;
   149→      p.shoulderL = 26 + Math.sin(t * 8 - 0.7) * 5;
   150→      p.elbowL = -55;
   151→      p.shoulderR = 26 + Math.sin(t * 8 - 0.7) * 5;
   152→      p.elbowR = -55;
   153→      p.mouthOpen = 0.85 + Math.sin(t * 8) * 0.15;
   154→      p.bodyLean = Math.sin(t * 4) * 3;
   155→      p.headTilt = Math.sin(t * 4 - 0.5) * 11; // head lags the torso
   156→      applyExpression(p, 'laugh', 1); // happy eyes, raised brows, mouth floor
   157→      break;
   158→    }
   159→    case 'point': {
   160→      // overshoot-and-settle: arm snaps out (~0.35s), overshoots, settles
   161→      const ext = reach(t, 0.35, 1.6);
   162→      p.shoulderR = 8 + ext * 80 + Math.sin(t * 2.4) * 2;
   163→      p.elbowR = -4 - ext * 4 + Math.sin(t * 2.4 - 0.4) * 3;
   164→      p.shoulderL = 8;
   165→      p.bodyLean = 5;
   166→      // head leads the gesture slightly, then recovers onto the target
   167→      p.headTilt = 2 + 3 * Math.exp(-t * 2.5);
   168→      p.mouthOpen = speaking ? 0.15 + 0.75 * Math.abs(Math.sin(t * 12.5)) : 0.1;
   169→      p.browState = 'raised';
   170→      p.fist = false; // pointing finger, not a fist
   171→      if (blinking(t)) p.eyeState = 'blink';
   172→      p.expression = 'smile';
   173→      break;
   174→    }
   175→    case 'angry': {
   176→      p.shoulderL = 62;
   177→      p.elbowL = -118;
   178→      p.shoulderR = 62;
   179→      p.elbowR = -118;
   180→      p.yOff = Math.abs(Math.sin(t * 22)) * 1.6;
   181→      p.xOff = Math.sin(t * 34) * 0.5; // trembling fury
   182→      p.headNod = Math.sin(t * 10 - 0.6) * 3; // head shake lags the tremble
   183→      p.mouthOpen = speaking ? 0.2 + 0.65 * Math.abs(Math.sin(t * 14)) : 0.12;
   184→      p.fist = true; // clenched fists
   185→      applyExpression(p, 'angry', 1); // furious face + angerVein + steam
   186→      break;
   187→    }
   188→    case 'jump': {
   189→      const cyc = (t % 1.15) / 1.15;
   190→      // anticipation → launch → air → land (squash & stretch)
   191→      let arc = 0;
   192→      let landK = -1; // >= 0 once the landing follow-through starts
   193→      if (cyc < 0.18) {
   194→        // anticipation crouch
   195→        const k = cyc / 0.18;
   196→        arc = -k * 0.06;
   197→        p.squash = k * 0.7;
   198→      } else if (cyc < 0.78) {
   199→        arc = Math.sin(((cyc - 0.18) / 0.6) * Math.PI);
   200→        p.squash = -0.45 * arc;
   201→        p.fist = true; // fists clenched in the air
   202→      } else {
   203→        // landing impact + a second tiny rebound in the squash
   204→        landK = (cyc - 0.78) / 0.22;
   205→        p.squash =
   206→          Math.sin(landK * Math.PI) * 0.85 +
   207→          0.18 * Math.min(1, landK * 6) * settle(landK, 10, 4);
   208→      }
   209→      p.yOff = Math.max(0, arc) * 58;
   210→      p.hipL = -18 - arc * 16;
   211→      p.hipR = -14 - arc * 14;
   212→      p.kneeL = 20 + arc * 30;
   213→      p.kneeR = 16 + arc * 28;
   214→      if (landK >= 0) {
   215→        // follow-through: arms overshoot downward, then damped settle back up
   216→        const ft = Math.min(1, landK * 5) * settle(landK, 9, 4);
   217→        p.shoulderL = 150 - ft * 36;
   218→        p.shoulderR = 150 - ft * 36;
   219→        p.elbowL = -8 - ft * 14;
   220→        p.elbowR = -8 - ft * 14;
   221→        // head bobs down with a short delay, then recovers
   222→        p.headNod =
   223→          -12 * Math.min(1, Math.max(0, landK - 0.12) * 3) * Math.exp(-landK * 2.5);
   224→      } else {
   225→        p.shoulderL = 150 + arc * 25;
   226→        p.shoulderR = 150 + arc * 25;
   227→        p.elbowL = -8;
   228→        p.elbowR = -8;
   229→      }
   230→      p.expression = 'smile';
   231→      p.mouthOpen = 0.35 + arc * 0.4;
   232→      p.eyeState = 'wide';
   233→      p.browState = 'raised';
   234→      break;
   235→    }
   236→    case 'nod': {
   237→      // overlapping body: torso pitch trails the head nod by ~0.12s
   238→      p.headNod = Math.sin(t * 6.5) * 15;
   239→      p.bodyLean = Math.sin(t * 6.5 - 0.75) * 3;
   240→      p.yOff = Math.sin(t * 2) * 1.5;
   241→      p.shoulderL = 4; // shoulders stay relaxed
   242→      p.shoulderR = 4;
   243→      p.mouthOpen = speaking ? 0.15 + 0.6 * Math.abs(Math.sin(t * 11)) : 0.1;
   244→      p.expression = 'smile';
   245→      if (blinking(t)) p.eyeState = 'blink';
   246→      break;
   247→    }
   248→    case 'cry': {
   249→      // heaving sob: body heaves, shoulders hug with a delay (overlapping)
   250→      p.yOff = Math.abs(Math.sin(t * 3.2)) * 5;
   251→      p.headTilt = Math.sin(t * 3.2 - 0.4) * 6;
   252→      p.shoulderL = 70 + Math.sin(t * 3.2 - 0.5) * 7;
   253→      p.elbowL = -95;
   254→      p.shoulderR = 70 + Math.sin(t * 3.2 - 0.9) * 7;
   255→      p.elbowR = -95;
   256→      p.mouthOpen = 0.3 + 0.25 * Math.abs(Math.sin(t * 7));
   257→      applyExpression(p, 'cry'); // closed eyes, sad brows, tears, mouth floor
   258→      break;
   259→    }
   260→    case 'run': {
   261→      const ph = t * TAU * 3.1;
   262→      p.hipL = Math.sin(ph) * 42;
   263→      p.hipR = Math.sin(ph + Math.PI) * 42;
   264→      p.kneeL = Math.max(0, Math.sin(ph + 1.1)) * 55;
   265→      p.kneeR = Math.max(0, Math.sin(ph + Math.PI + 1.1)) * 55;
   266→      // big arm swing, elbows trailing by ~0.6 rad (overlapping action)
   267→      p.shoulderL = 8 + Math.sin(ph + Math.PI) * 42;
   268→      p.shoulderR = 8 + Math.sin(ph) * 42;
   269→      p.elbowL = -68 + lagSin(ph + Math.PI, 0.6) * 12;
   270→      p.elbowR = -68 + lagSin(ph, 0.6) * 12;
   271→      p.yOff = Math.abs(Math.sin(ph)) * 7; // strong flight bounce
   272→      p.xOff = Math.sin(ph - 0.5) * 2.4; // sway trails the feet
   273→      p.bodyLean = 10 + Math.sin(ph) * 2.5; // forward lean pulses with stride
   274→      p.headNod = Math.sin(ph * 2 - 0.55) * 2.8; // delayed counter-bob
   275→      p.fist = true; // sprinting fists
   276→      p.expression = 'neutral';
   277→      p.mouthOpen = speaking ? 0.2 + 0.7 * Math.abs(Math.sin(t * 12)) : 0.18;
   278→      if (blinking(t)) p.eyeState = 'blink';
   279→      break;
   280→    }
   281→    case 'slap': {
   282→      // 1.1s cycle: windup → whoosh → follow-through settle
   283→      const cyc = (t % 1.1) / 1.1;
   284→      let swing: number;
   285→      let elbow: number;
   286→      if (cyc < 0.42) {
   287→        // anticipation: arm winds back
   288→        const k = cyc / 0.42;
   289→        swing = -k * 46;
   290→        elbow = -k * 66;
   291→        p.bodyLean = -k * 7;
   292→      } else if (cyc < 0.62) {
   293→        // fast strike
   294→        const k = (cyc - 0.42) / 0.2;
   295→        swing = -46 + k * 190;
   296→        elbow = -66 + k * 58;
   297→        p.bodyLean = -7 + k * 16;
   298→        p.squash = Math.sin(k * Math.PI) * 0.3;
   299→      } else {
   300→        // follow-through: damped settle instead of a static hold
   301→        const k = (cyc - 0.62) / 0.38;
   302→        const ramp = Math.min(1, k * 8);
   303→        swing = 144 * (1 - k * 0.25) + ramp * settle(k, 11, 4.5) * 9;
   304→        elbow = -8 + ramp * settle(Math.max(0, k - 0.06) / 0.94, 11, 4.5) * 6;
   305→        p.bodyLean = 9 * (1 - k * 0.4);
   306→      }
   307→      p.shoulderR = 62 + swing;
   308→      p.elbowR = elbow;
   309→      p.shoulderL = -14;
   310→      p.elbowL = -34;
   311→      // head: shake during windup/strike, recoil snap with delay on impact
   312→      if (cyc < 0.62) {
   313→        p.headNod = Math.sin(cyc * 18) * 3;
   314→      } else {
   315→        const k = (cyc - 0.62) / 0.38;
   316→        const ramp = Math.min(1, k * 12);
   317→        p.headNod = ramp * (-4 * Math.exp(-k * 3.2) + 3 * settle(k, 12, 5));
   318→      }
   319→      p.browState = 'angry';
   320→      p.expression = 'angry';
   321→      p.eyeState = 'wide';
   322→      p.fist = true; // clenched fist
   323→      p.mouthOpen = speaking ? 0.25 + 0.6 * Math.abs(Math.sin(t * 13)) : 0.2;
   324→      break;
   325→    }
   326→    case 'bow': {
   327→      const cyc = (t % 2.4) / 2.4;
   328→      // anticipation: a tiny rise before folding down (first 8%)
   329→      const antic = cyc < 0.08 ? Math.sin((cyc / 0.08) * Math.PI) * 2 : 0;
   330→      // soft settle wobble around the deep point of the bow
   331→      const wob =
   332→        cyc > 0.31
   333→          ? settle((cyc - 0.31) / 0.45, 6, 3) * 2.6 * Math.min(1, (cyc - 0.31) * 12)
   334→          : 0;
   335→      // lean down, hold with the wobble, rise
   336→      const lean = Math.sin(Math.min(1, cyc * 1.6) * Math.PI) * 34 + wob;
   337→      p.bodyLean = lean;
   338→      p.headNod = lean * 0.55;
   339→      p.shoulderL = 18 + lean * 1.1;
   340→      p.shoulderR = 18 + lean * 1.1;
   341→      p.elbowL = -58 - lean * 0.5;
   342→      p.elbowR = -58 - lean * 0.5;
   343→      p.yOff = antic - lean * 0.35;
   344→      p.expression = 'smile';
   345→      p.mouthOpen = speaking ? 0.15 + 0.6 * Math.abs(Math.sin(t * 12)) : 0.08;
   346→      // eyes close at the full depth of the bow (respect)
   347→      if (lean > 26) p.eyeState = 'closed';
   348→      else if (blinking(t)) p.eyeState = 'blink';
   349→      break;
   350→    }
   351→    case 'think': {
   352→      // head-scratch secondary action: forearm draws small slow circles
   353→      p.headTilt = 9 + Math.sin(t * 1.6) * 2.5;
   354→      p.shoulderR = 34 + Math.sin(t * 2.2 - 1.1) * 3; // slight counter to elbow
   355→      p.elbowR = -122 + Math.sin(t * 2.2) * 6;
   356→      p.shoulderL = 6;
   357→      p.elbowL = -10;
   358→      p.yOff = Math.sin(t * 1.9) * 1.6;
   359→      p.browState = 'raised';
   360→      p.expression = 'neutral';
   361→      p.mouthOpen = speaking ? 0.12 + 0.55 * Math.abs(Math.sin(t * 11)) : 0.06;
   362→      if (blinking(t)) p.eyeState = 'blink';
   363→      break;
   364→    }
   365→    case 'surprise': {
   366→      const cyc = (t % 1.6) / 1.6;
   367→      const jolt = Math.exp(-cyc * 5) * Math.sin(cyc * 22);
   368→      // a soft settle wobble rides in after the initial jolt
   369→      const wob = Math.min(1, cyc * 12) * settle(cyc, 8, 3.2) * 3.2;
   370→      p.yOff = Math.max(0, jolt) * 26 + wob;
   371→      p.squash = -Math.max(0, jolt) * 0.35;
   372→      // arms overshoot upward, then settle with the wobble
   373→      const armWob = Math.min(1, cyc * 10) * settle(cyc, 9, 3.5) * 8;
   374→      p.shoulderL = 148 + jolt * 18 + armWob;
   375→      p.shoulderR = 148 + jolt * 18 + armWob;
   376→      p.elbowL = -12;
   377→      p.elbowR = -12;
   378→      p.headTilt = jolt * 6;
   379→      p.mouthOpen = 0.55 + Math.abs(jolt) * 0.3;
   380→      applyExpression(p, 'surprised'); // wide eyes + shockLines FX
   381→      break;
   382→    }
   383→    case 'dance': {
   384→      const ph = t * TAU * 1.35; // body rhythm 1.35 Hz
   385→      const armPh = ph - 0.35; // arms lag the body (overlapping action)
   386→      p.xOff = Math.sin(ph) * 6;
   387→      p.yOff = Math.abs(Math.sin(ph * 2)) * 4.5;
   388→      p.bodyLean = Math.sin(ph) * 7;
   389→      p.headTilt = Math.sin(ph - 0.8) * 9; // head lags even more
   390→      p.headNod = Math.sin(ph * 2 - 0.5) * 3;
   391→      p.shoulderL = 60 + Math.sin(armPh) * 95;
   392→      p.shoulderR = 60 - Math.sin(armPh) * 95;
   393→      p.elbowL = -34 + Math.cos(armPh - 0.35) * 22;
   394→      p.elbowR = -34 - Math.cos(armPh - 0.35) * 22;
   395→      // feet: hips alternate, knees bounce on the double beat
   396→      p.hipL = Math.sin(ph) * 14;
   397→      p.hipR = Math.sin(ph + Math.PI) * 14;
   398→      p.kneeL = Math.max(0, Math.sin(ph * 2 - 0.4)) * 16;
   399→      p.kneeR = Math.max(0, Math.sin(ph * 2 - 0.4 + Math.PI)) * 16;
   400→      p.eyeState = 'happy';
   401→      p.browState = 'raised';
   402→      p.expression = 'laugh';
   403→      p.fx = ['music'];
   404→      p.mouthOpen = 0.45 + Math.abs(Math.sin(ph * 2)) * 0.3;
   405→      break;
   406→    }
   407→    case 'fall': {
   408→      // comedic stumble-back cycle: windmill wobble → tip → sit → recover
   409→      const cyc = (t % 2.2) / 2.2;
   410→      if (cyc < 0.3) {
   411→        const k = cyc / 0.3;
   412→        p.bodyLean = -k * 16;
   413→        // bigger anticipation: panic windmill, shoulders rotate full-ish circles
   414→        p.shoulderL = 40 + k * 50 + Math.sin(t * 18) * 35;
   415→        p.shoulderR = 40 + k * 50 + Math.sin(t * 18 + Math.PI) * 35;
   416→        p.elbowL = -100 + Math.sin(t * 18 - 0.5) * 12;
   417→        p.elbowR = -100 + Math.sin(t * 18 - 0.5 + Math.PI) * 12;
   418→        p.eyeState = 'wide';
   419→      } else if (cyc < 0.55) {
   420→        const k = (cyc - 0.3) / 0.25;
   421→        p.bodyLean = -16 - k * 58;
   422→        p.yOff = -k * 34;
   423→        // arms flail with lag while tipping over
   424→        p.shoulderL = 90 + k * 80 + Math.sin(t * 18 - 0.4) * 18;
   425→        p.shoulderR = 90 + k * 80 + Math.sin(t * 18 - 0.9) * 18;
   426→        p.elbowL = -60 + Math.sin(t * 18 - 0.6) * 20;
   427→        p.elbowR = -60 + Math.sin(t * 18 - 1.1) * 20;
   428→        p.hipL = -30 * k;
   429→        p.hipR = -26 * k;
   430→        p.kneeL = 40 * k;
   431→        p.kneeR = 34 * k;
   432→        p.expression = 'surprised';
   433→        p.browState = 'raised';
   434→        p.eyeState = 'wide';
   435→        p.mouthOpen = 0.7;
   436→      } else if (cyc < 0.85) {
   437→        // sitting on ground
   438→        p.bodyLean = -74;
   439→        p.yOff = -34;
   440→        p.shoulderL = 150;
   441→        p.shoulderR = 150;
   442→        p.elbowL = -40;
   443→        p.elbowR = -40;
   444→        p.hipL = -58;
   445→        p.hipR = -52;
   446→        p.kneeL = 58;
   447→        p.kneeR = 50;
   448→        p.expression = 'surprised';
   449→        p.browState = 'raised';
   450→        p.mouthOpen = 0.5;
   451→        p.tears = false;
   452→      } else {
   453→        const k = (cyc - 0.85) / 0.15;
   454→        // recovery with a soft settle wobble (follow-through)
   455→        const wob = Math.min(1, k * 4) * settle(k, 9, 3.5) * 6;
   456→        p.bodyLean = -74 * (1 - k) + wob * 0.4;
   457→        p.yOff = -34 * (1 - k);
   458→        p.shoulderL = 150 - k * 120 + wob;
   459→        p.shoulderR = 150 - k * 120 + wob;
   460→        p.elbowL = -40 + k * 20;
   461→        p.elbowR = -40 + k * 20;
   462→        p.hipL = -58 * (1 - k);
   463→        p.hipR = -52 * (1 - k);
   464→        p.kneeL = 58 * (1 - k);
   465→        p.kneeR = 50 * (1 - k);
   466→        p.expression = 'frown';
   467→        p.browState = 'sad';
   468→      }
   469→      break;
   470→    }
   471→    case 'push': {
   472→      const ph = Math.sin(t * 5.2);
   473→      // strain: body leads, elbows follow with a 0.4 rad lag
   474→      p.bodyLean = 12 + ph * 3;
   475→      p.shoulderL = 84 + Math.sin(t * 5.2 - 0.2) * 8;

   488→      p.mouthOpen = speaking ? 0.2 + 0.6 * Math.abs(Math.sin(t * 12)) : 0.15;
   489→      if (blinking(t)) p.eyeState = 'blink';

   491→    }
   492→    case 'kick': {
   493→      // 1.3s cycle: knee windup → leg whip → settle (Gopal-style comedy kick)
   494→      const cyc = (t % 1.3) / 1.3;
   495→      let hip: number;
   496→      let knee: number;
   497→      let lean: number;

   527→      p.shoulderR = 30 + hip * 0.5;

   532→      p.expression = 'angry';

   534→      p.mouthOpen = speaking ? 0.25 + 0.6 * Math.abs(Math.sin(t * 13)) : 0.3;

   536→    }
   537→    case 'shiver': {
   538→      // frightened trembling: high-freq jitter + chattering teeth
   539→      const j = Math.sin(t * 26) * 1.6;

   546→      p.elbowL = -108 + Math.sin(t * 24 - 0.5) * 4;
   547→      p.shoulderR = 46 + Math.sin(t * 24 + 1.4) * 3;

   549→      p.kneeL = 8;
   550→      p.kneeR = 6; // knees knocking
   551→      p.mouthOpen = 0.1 + Math.abs(Math.sin(t * 21)) * 0.14; // b-b-b-chat chatter
   552→      p.eyeState = 'wide';
   553→      p.browState = 'sad';
   554→      p.expression = 'frown';

   557→    }
   558→    case 'eat': {

   577→    }

   656→export function easeInOut(x: number): number {
   657→  return x < 0.5 ? 2 * x * x : 1 - Math.pow(-2 * x + 2, 2) / 2;
   658→}

   660→/** Movement-quality easings for beat paths (walk/run targets). */
   661→export function easingByName(name: EasingName): (x: number) => number {
   662→  switch (name) {
   663→    case 'linear':
   664→      return (x) => x;
   665→    case 'easeIn':
   666→      return (x) => x * x;
   667→    case 'easeOut':
   668→      return (x) => 1 - (1 - x) * (1 - x);
   669→    case 'backOut':
   670→      return (x) => {
   671→        const c1 = 1.70158;
   672→        const c3 = c1 + 1;
   673→        return 1 + c3 * Math.pow(x - 1, 3) + c1 * Math.pow(x - 1, 2);
   674→      };
   675→    case 'elasticOut':
   676→      return (x) => {
   677→        if (x === 0 || x === 1) return x;
   678→        const c4 = (2 * Math.PI) / 3;
   679→        return Math.pow(2, -10 * x) * Math.sin((x * 10 - 0.75) * c4) + 1;
   680→      };
   681→    case 'easeInOut':
   682→    default:
   683→      return easeInOut;
   684→  }
   685→}
