    91→function pickMime(): { mime: string; ext: string } {
    92→  const candidates = [
    93→    "video/mp4;codecs=avc1.42E01E,mp4a.40.2",
    94→    "video/mp4",
    95→    "video/webm;codecs=vp9,opus",
    96→    "video/webm;codecs=vp8,opus",
    97→    "video/webm",
    98→  ];
    99→  if (typeof MediaRecorder === "undefined") return { mime: "", ext: "webm" };
   100→  for (const m of candidates) {
   101→    if (MediaRecorder.isTypeSupported(m)) {
   102→      return { mime: m, ext: m.startsWith("video/mp4") ? "mp4" : "webm" };
   103→    }
   104→  }
   105→  return { mime: "", ext: "webm" };
   106→}
