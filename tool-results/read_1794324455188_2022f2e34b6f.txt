    90→function pickMime(): { mime: string; ext: string } {
    91→  const candidates = [
    92→    "video/mp4;codecs=avc1.42E01E,mp4a.40.2",
    93→    "video/mp4",
    94→    "video/webm;codecs=vp9,opus",
    95→    "video/webm;codecs=vp8,opus",
    96→    "video/webm",
    97→  ];
    98→  if (typeof MediaRecorder === "undefined") return { mime: "", ext: "webm" };
    99→  for (const m of candidates) {
   100→    if (MediaRecorder.isTypeSupported(m)) {
   101→      return { mime: m, ext: m.startsWith("video/mp4") ? "mp4" : "webm" };
   102→    }
   103→  }
   104→  return { mime: "", ext: "webm" };
   105→}
