    40→    40→  let acc = 0;

    45→    45→function wrapText(
    46→    46→  ctx: CanvasRenderingContext2D,
    47→    47→  text: string,
    48→    48→  maxWidth: number,
    49→    49→): string[] {
    50→    50→  const words = text.split(/\s+/).filter(Boolean);
    51→    51→  const lines: string[] = [];
    52→    52→  let cur = '';
    53→    53→  for (const w of words) {
    54→    54→    const test = cur ? cur + ' ' + w : w;
    55→    55→    if (ctx.measureText(test).width > maxWidth && cur) {
    56→    56→      lines.push(cur);
    57→    57→      cur = w;
    58→    58→    } else {
    59→    59→      cur = test;
    60→    60→    }
    61→    61→  }
    62→    62→  if (cur) lines.push(cur);
    63→    63→  return lines;
    64→    64→}

   131→   131→  setScene(scene: Scene, designs: CharacterDesign[]) {
   132→   132→    this.scene = scene;
   133→   133→    this.designs = new Map(designs.map((d) => [d.id, d]));
   134→   134→    // keep the editable walk-flag if its beat still exists and is a walk/run
   135→   135→    if (this.editBeatId) {
   136→   136→      const b = scene.beats.find((x) => x.id === this.editBeatId);
   137→   137→      if (!b || (b.action !== 'walk' && b.action !== 'run')) this.editBeatId = null;
   138→   138→    }
   139→   139→    if (this.scene.beats.length === 0) {
   140→   140→      this.sceneTime = 0;
   141→   141→      this.activeIds.clear();
   142→   142→      this.callbacks.onTime?.(0, 0);
   143→   143→      return;
   144→   144→    }
   145→   145→    // preserve the playhead position across scene edits (never rewind!)
   146→   146→    const total = this.totalDuration();
   147→   147→    const clamped = Math.min(this.sceneTime, Math.max(0, total - 0.001));
   148→   148→    if (Math.abs(clamped - this.sceneTime) > 0.001) {
   149→   149→      this.seek(clamped);
   150→   150→    } else {
   151→   151→      // re-sync active windows for the new layout (no audio restart)
   152→   152→      const active = this.activeAt(this.layout(), this.sceneTime);
   153→   153→      this.activeIds = new Set(active.map((w) => w.beat.id));
   154→   154→      this.callbacks.onTime?.(this.sceneTime, active.length);
   155→   155→    }
   156→   156→  }

   203→   203→  // ---------------- timeline layout ----------------
   204→   204→
   205→   205→  /** All beat windows sorted by start time (parallel-safe). */
   206→   206→  private layout(): BeatWindow[] {
   207→   207→    if (!this.scene) return [];
   208→   208→    const beats = this.scene.beats;
   209→   209→    const wins = beats.map((b, i) => {
   210→   210→      const start =
   211→   211→        typeof b.start === 'number' && b.start >= 0 ? b.start : seqStart(beats, i);
   212→   212→      const dur = Math.max(0.3, this.durations[b.id] ?? b.duration);
   213→   213→      return { beat: b, idx: i, start, end: start + dur } as BeatWindow;
   214→   214→    });
   215→   215→    wins.sort((a, b) => a.start - b.start || a.idx - b.idx);
   216→   216→    return wins;
   217→   217→  }
   218→   218→
   219→   219→  private totalDuration(): number {
   220→   220→    let total = 0;
   221→   221→    for (const w of this.layout()) total = Math.max(total, w.end);
   222→   222→    return total;
   223→   223→  }
   224→   224→
   225→   225→  private activeAt(wins: BeatWindow[], t: number): BeatWindow[] {
   226→   226→    return wins.filter((w) => w.start <= t && t < w.end);
   227→   227→  }
   228→   228→
   229→   229→  /**
   230→   230→   * Where does this character stand at scene time `t`?
   231→   231→   * Deterministic walk-chain: x = the destination of the latest walk that
   232→   232→   * already finished before t (so chained walks never teleport).
   233→   233→   */
   234→   234→  private charXAt(uid: string, t: number, placed: PlacedCharacter): number {
   235→   235→    let x = (placed.x / 100) * STAGE_W;
   236→   236→    let best: BeatWindow | null = null;
   237→   237→    for (const w of this.layout()) {
   238→   238→      if (
   239→   239→        w.beat.actorUid === uid &&
   240→   240→        (w.beat.action === 'walk' || w.beat.action === 'run') &&
   241→   241→        w.beat.moveTo != null &&
   242→   242→        w.end <= t + 1e-6
   243→   243→      ) {
   244→   244→        if (!best || w.end > best.end) best = w;
   245→   245→      }
   246→   246→    }
   247→   247→    if (best && best.beat.moveTo != null) x = (best.beat.moveTo / 100) * STAGE_W;
   248→   248→    return x;
   249→   249→  }
   250→   250→
   251→   251→  // ---------------- transport ----------------
   252→   252→
   253→   253→  /** Jump to an absolute scene time (works while playing or paused). */
   254→   254→  seek(t: number) {
   255→   255→    const wins = this.layout();
   256→   256→    if (wins.length === 0) return;
   257→   257→    const total = this.totalDuration();
   258→   258→    const clamped = clamp(t, 0, Math.max(0, total - 0.001));
   259→   259→    this.sceneTime = clamped;
   260→   260→    const active = this.activeAt(wins, clamped);
   261→   261→    const newIds = new Set(active.map((w) => w.beat.id));
   262→   262→
   263→   263→    if (this.playing) {
   264→   264→      for (const id of this.activeIds) {
   265→   265→        if (!newIds.has(id)) {
   266→   266→          const w = wins.find((x) => x.beat.id === id);
   267→   267→          if (w) this.callbacks.onBeatEnd?.(w.beat);
   268→   268→        }
   269→   269→      }
   270→   270→      for (const w of active) {
   271→   271→        if (!this.activeIds.has(w.beat.id)) {
   272→   272→          this.callbacks.onBeatStart?.(w.beat, Math.max(0, clamped - w.start));
   273→   273→        }
   274→   274→      }
   275→   275→    }
   276→   276→    this.activeIds = newIds;
   277→   277→    this.callbacks.onTime?.(clamped, active.length);
   278→   278→  }
   279→   279→
   280→   280→  play(): boolean {
   281→   281→    const wins = this.layout();
   282→   282→    if (wins.length === 0) return false;
   283→   283→    if (this.sceneTime >= this.totalDuration() - 0.02) this.sceneTime = 0;
   284→   284→    const active = this.activeAt(wins, this.sceneTime);
   285→   285→    this.playing = true;
   286→   286→    this.activeIds = new Set(active.map((w) => w.beat.id));
   287→   287→    for (const w of active) {
   288→   288→      this.callbacks.onBeatStart?.(w.beat, Math.max(0, this.sceneTime - w.start));
   289→   289→    }
   290→   290→    this.callbacks.onTime?.(this.sceneTime, active.length);
   291→   291→    return true;
   292→   292→  }
   293→   293→
   294→   294→  pause() {
   295→   295→    if (!this.playing) return;
   296→   296→    this.playing = false;
   297→   297→    const wins = this.layout();
   298→   298→    for (const id of this.activeIds) {
   299→   299→      const w = wins.find((x) => x.beat.id === id);
   300→   300→      if (w) this.callbacks.onBeatEnd?.(w.beat);
   301→   301→    }
   302→   302→    this.activeIds = new Set();
   303→   303→  }
   304→   304→
   305→   305→  stop() {
   306→   306→    this.pause();
   307→   307→    this.sceneTime = 0;
   308→   308→    this.callbacks.onTime?.(0, 0);
   309→   309→  }

   323→   323→  /** x of the editable walk-target flag, or null when not shown */
   324→   324→  private flagX(): number | null {
   325→   325→    if (this.playing || !this.editBeatId || !this.scene) return null;
   326→   326→    const beat = this.scene.beats.find((b) => b.id === this.editBeatId);
   327→   327→    if (!beat) return null;
   328→   328→    if (beat.action !== 'walk' && beat.action !== 'run') return null;
   329→   329→    if (beat.moveTo == null) return null;
   330→   330→    return (beat.moveTo / 100) * STAGE_W;
   331→   331→  }
   332→   332→
   333→   333→  private overFlag(x: number, y: number): boolean {
   334→   334→    const fx = this.flagX();
   335→   335→    if (fx == null) return false;
   336→   336→    return Math.abs(x - fx) <= 30 && y >= GROUND_Y - 92 && y <= GROUND_Y + 18;
   337→   337→  }
   338→   338→
   339→   339→  private hitCharacter(x: number, y: number): PlacedCharacter | null {
   340→   340→    if (!this.scene) return null;
   341→   341→    for (let i = this.scene.characters.length - 1; i >= 0; i--) {
   342→   342→      const c = this.scene.characters[i];
   343→   343→      const cx = (c.x / 100) * STAGE_W;
   344→   344→      const s = c.scale;
   345→   345→      if (Math.abs(x - cx) < 60 * s && y > GROUND_Y - 245 * s && y < GROUND_Y + 14) {
   346→   346→        return c;
   347→   347→      }
   348→   348→    }
   349→   349→    return null;
   350→   350→  }

   353→   353→    this.canvas.addEventListener('pointerdown', this.onPointerDown);
   354→   354→    this.canvas.addEventListener('pointermove', this.onPointerMove);
   355→   355→    this.canvas.addEventListener('pointerup', this.onPointerUp);
   356→   356→    this.canvas.addEventListener('pointercancel', this.onPointerUp);
   357→   357→  }
   358→   358→
   359→   359→  private detachPointer() {
   360→   360→    this.canvas.removeEventListener('pointerdown', this.onPointerDown);
   361→   361→    this.canvas.removeEventListener('pointermove', this.onPointerMove);
   362→   362→    this.canvas.removeEventListener('pointerup', this.onPointerUp);
   363→   363→    this.canvas.removeEventListener('pointercancel', this.onPointerUp);
   364→   364→  }
   365→   365→
   366→   366→  private onPointerDown = (e: PointerEvent) => {
   367→   367→    const x = this.canvasX(e);
   368→   368→    const y = this.canvasY(e);
   369→   369→    // walk-target flag has priority
   370→   370→    if (this.overFlag(x, y)) {
   371→   371→      this.flagDrag = true;
   372→   372→      this.canvas.setPointerCapture(e.pointerId);
   373→   373→      this.interactions.onDragTarget?.(clamp((x / STAGE_W) * 100, 4, 96));
   374→   374→      this.canvas.style.cursor = 'ew-resize';
   375→   375→      return;
   376→   376→    }
   377→   377→    const hit = this.hitCharacter(x, y);
   378→   378→    if (hit) {
   379→   379→      this.dragUid = hit.uid;
   380→   380→      this.dragMoved = false;
   381→   381→      this.selectedUid = hit.uid;
   382→   382→      this.canvas.setPointerCapture(e.pointerId);
   383→   383→      this.interactions.onPick?.(hit.uid);
   384→   384→    } else {
   385→   385→      this.selectedUid = null;
   386→   386→      this.interactions.onPick?.(null);
   387→   387→    }
   388→   388→  };
   389→   389→
   390→   390→  private onPointerMove = (e: PointerEvent) => {
   391→   391→    const x = this.canvasX(e);
   392→   392→    const y = this.canvasY(e);
   393→   393→    if (this.flagDrag) {
   394→   394→      this.interactions.onDragTarget?.(clamp((x / STAGE_W) * 100, 4, 96));
   395→   395→      this.canvas.style.cursor = 'ew-resize';
   396→   396→      return;
   397→   397→    }
   398→   398→    if (this.dragUid) {
   399→   399→      this.dragMoved = true;
   400→   400→      const pct = clamp((x / STAGE_W) * 100, 4, 96);
   401→   401→      this.interactions.onDrag?.(this.dragUid, pct);
   402→   402→      this.canvas.style.cursor = 'grabbing';
   403→   403→      return;
   404→   404→    }
   405→   405→    if (this.overFlag(x, y)) {
   406→   406→      this.canvas.style.cursor = 'ew-resize';
   407→   407→      return;
   408→   408→    }
   409→   409→    const hit = this.hitCharacter(x, y);
   410→   410→    this.canvas.style.cursor = hit ? 'grab' : 'default';
   411→   411→  };
   412→   412→
   413→   413→  private onPointerUp = (e: PointerEvent) => {
   414→   414→    if (this.flagDrag) {
   415→   415→      try {
   416→   416→        this.canvas.releasePointerCapture(e.pointerId);
   417→   417→      } catch {
   418→   418→        /* ignore */
   419→   419→      }
   420→   420→      this.flagDrag = false;
   421→   421→      const pct = clamp((this.canvasX(e) / STAGE_W) * 100, 4, 96);
   422→   422→      this.interactions.onDragTargetEnd?.(pct);
   423→   423→      this.canvas.style.cursor = 'default';
   424→   424→      return;
   425→   425→    }
   426→   426→    if (this.dragUid) {
   427→   427→      try {
   428→   428→        this.canvas.releasePointerCapture(e.pointerId);
   429→   429→      } catch {
   430→   430→        /* ignore */
   431→   431→      }
   432→   432→      const pct = clamp((this.canvasX(e) / STAGE_W) * 100, 4, 96);
   433→   433→      if (this.dragMoved) this.interactions.onDragEnd?.(this.dragUid, pct);
   434→   434→      this.dragUid = null;
   435→   435→      this.canvas.style.cursor = 'grab';
   436→   436→    }
   437→   437→  };

   439→   439→  // ---------------- main loop ----------------
   440→   440→
   441→   441→  private loop = (ts: number) => {
   442→   442→    this.raf = requestAnimationFrame(this.loop);
   443→   443→    const dt = Math.min(0.1, (ts - this.lastTs) / 1000);
   444→   444→    this.lastTs = ts;
   445→   445→    this.animT += dt * this.rate;
   446→   446→
   447→   447→    if (this.playing && this.scene && this.scene.beats.length > 0) {
   448→   448→      this.sceneTime += dt * this.rate;
   449→   449→      const wins = this.layout();
   450→   450→      const total = this.totalDuration();
   451→   451→      if (this.sceneTime >= total) {
   452→   452→        this.sceneTime = total;
   453→   453→        const stillActive = wins.filter((w) => this.activeIds.has(w.beat.id));
   454→   454→        for (const w of stillActive) this.callbacks.onBeatEnd?.(w.beat);
   455→   455→        this.activeIds.clear();
   456→   456→        this.playing = false;
   457→   457→        this.callbacks.onTime?.(this.sceneTime, 0);
   458→   458→        this.callbacks.onFinish?.();
   459→   459→      } else {
   460→   460→        const active = this.activeAt(wins, this.sceneTime);
   461→   461→        const newIds = new Set(active.map((w) => w.beat.id));
   462→   462→        for (const id of this.activeIds) {
   463→   463→          if (!newIds.has(id)) {
   464→   464→            const w = wins.find((x) => x.beat.id === id);
   465→   465→            if (w) this.callbacks.onBeatEnd?.(w.beat);
   466→   466→          }
   467→   467→        }
   468→   468→        for (const w of active) {
   469→   469→          if (!this.activeIds.has(w.beat.id)) {
   470→   470→            this.callbacks.onBeatStart?.(w.beat, Math.max(0, this.sceneTime - w.start));
   471→   471→          }
   472→   472→        }
   473→   473→        this.activeIds = newIds;
   474→   474→        this.callbacks.onTime?.(this.sceneTime, active.length);
   475→   475→      }
   476→   476→    }
   477→   477→
   478→   478→    this.render();
   479→   479→  };
   480→   480→
   481→   481→  // ---------------- rendering ----------------
   482→   482→
   483→   483→  private render() {
   484→   484→    const ctx = this.ctx;
   485→   485→    ctx.clearRect(0, 0, STAGE_W, STAGE_H);
   486→   486→
   487→   487→    if (!this.scene || this.scene.beats.length === 0) {
   488→   488→      // empty stage splash
   489→   489→      ctx.fillStyle = '#fff7ed';
   490→   490→      ctx.fillRect(0, 0, STAGE_W, STAGE_H);
   491→   491→      ctx.fillStyle = '#9a3412';
   492→   492→      ctx.font = '700 42px "Hind Siliguri", sans-serif';
   493→   493→      ctx.textAlign = 'center';
   494→   494→      ctx.textBaseline = 'middle';
   495→   495→      ctx.fillText('ঠাট্টা স্টুডিও', STAGE_W / 2, STAGE_H / 2 - 20);
   496→   496→      ctx.font = '500 26px "Hind Siliguri", sans-serif';
   497→   497→      ctx.fillStyle = '#c2410c';
   498→   498→      ctx.fillText(
   499→   499→        'ক্যারেক্টার বসিয়ে কার্টুন বানানো শুরু করুন',
   500→   500→        STAGE_W / 2,
   501→   501→        STAGE_H / 2 + 28,
   502→   502→      );
   503→   503→      return;
   504→   504→    }
   505→   505→
   506→   506→    // background
   507→   507→    try {
   508→   508→      ctx.drawImage(getBackgroundCanvas(this.scene.background), 0, 0);
   509→   509→    } catch {
   510→   510→      ctx.fillStyle = '#fde68a';
   511→   511→      ctx.fillRect(0, 0, STAGE_W, STAGE_H);
   512→   512→    }
   513→   513→
   514→   514→    // living background layer (clouds, fireflies, shimmer…)
   515→   515→    try {
   516→   516→      drawAmbient(ctx, this.scene.background, this.animT);
   517→   517→    } catch {
   518→   518→      /* ambient is decorative — never break the stage */
   519→   519→    }
   520→   520→
   521→   521→    const t = this.sceneTime;
   522→   522→    const inScene = this.playing || t > 0;
   523→   523→    const wins = this.layout();
   524→   524→    const active = inScene ? this.activeAt(wins, t) : [];
   525→   525→
   526→   526→    // one active window per actor (latest start wins)
   527→   527→    const activeByActor = new Map<string, BeatWindow>();
   528→   528→    for (const w of active) {
   529→   529→      const prev = activeByActor.get(w.beat.actorUid);
   530→   530→      if (!prev || w.start > prev.start) activeByActor.set(w.beat.actorUid, w);
   531→   531→    }
   532→   532→
   533→   533→    for (const placed of this.scene.characters) {
   534→   534→      const design = this.designs.get(placed.designId);
   535→   535→      if (!design) continue;
   536→   536→
   537→   537→      const w = inScene ? (activeByActor.get(placed.uid) ?? null) : null;
   538→   538→      let worldX = this.charXAt(placed.uid, inScene ? t : 0, placed);
   539→   539→      let flip = placed.flip;
   540→   540→      let pose = basePose();
   541→   541→      let localT = this.animT;
   542→   542→
   543→   543→      if (w) {
   544→   544→        const beat = w.beat;
   545→   545→        const beatElapsed = Math.max(0, t - w.start);
   546→   546→        const speed = clamp(beat.speed ?? 1, 0.25, 2.5);
   547→   547→        localT = beatElapsed * speed;
   548→   548→        pose = computePose(beat.action, localT, beat.text.trim().length > 0);
   549→   549→
   550→   550→        // ---- real viseme lip-sync: audio envelopes drive the mouth ----
   551→   551→        const env = this.envelopes[beat.id];
   552→   552→        const wideEnv = this.wideEnvelopes[beat.id];
   553→   553→        if (env && env.length > 0 && beat.text.trim().length > 0) {
   554→   554→          const fi = beatElapsed * 24;
   555→   555→          const i0 = Math.max(0, Math.min(env.length - 1, Math.floor(fi)));
   556→   556→          const i1 = Math.max(0, Math.min(env.length - 1, i0 + 1));
   557→   557→          const frac = fi - Math.floor(fi);
   558→   558→          const amp = lerp(env[i0], env[i1], frac);
   559→   559→          pose.mouthOpen = Math.max(0.06, Math.min(1, 0.12 + amp * 1.05));
   560→   560→          // “ee/ai” viseme widens the mouth horizontally when present
   561→   561→          if (wideEnv && wideEnv.length > 0) {
   562→   562→            const w0 = wideEnv[Math.max(0, Math.min(wideEnv.length - 1, Math.floor(fi)))];
   563→   563→            const w1 =
   564→   564→              wideEnv[Math.max(0, Math.min(wideEnv.length - 1, Math.floor(fi) + 1))] ?? w0;
   565→   565→            pose.mouthWide = 0.18 + lerp(w0, w1, frac) * 0.82;
   566→   566→          } else {
   567→   567→            pose.mouthWide = 0.25 + amp * 0.35;
   568→   568→          }
   569→   569→          // tiny speech head-bob for liveliness
   570→   570→          pose.headNod += (amp - 0.35) * 2.2;
   571→   571→        }
   572→   572→
   573→   573→        if (
   574→   574→          (beat.action === 'walk' || beat.action === 'run') &&
   575→   575→          beat.moveTo != null
   576→   576→        ) {
   577→   577→          const dur = Math.max(0.3, w.end - w.start);
   578→   578→          const progress = Math.min(1, beatElapsed / dur);
   579→   579→          const eased = easingByName(beat.easing ?? 'easeInOut')(progress);
   580→   580→          const startX = this.charXAt(placed.uid, w.start, placed);
   581→   581→          const targetX = (beat.moveTo / 100) * STAGE_W;
   582→   582→          worldX = lerp(startX, targetX, eased);
   583→   583→          flip = targetX < startX;
   584→   584→        }
   585→   585→      } else {
   586→   586→        pose = computePose('idle', localT, false);
   587→   587→      }
   588→   588→
   589→   589→      const s = placed.scale;
   590→   590→      ctx.save();
   591→   591→      ctx.translate(worldX, GROUND_Y);
   592→   592→      ctx.scale(flip ? -s : s, s);
   593→   593→      drawCharacter(ctx, design, pose, localT, placed.prop ?? 'none');
   594→   594→      ctx.restore();
   595→   595→
   596→   596→      // selection ring
   597→   597→      if (this.selectedUid === placed.uid) {
   598→   598→        ctx.save();
   599→   599→        ctx.strokeStyle = 'rgba(234, 88, 12, 0.85)';
   600→   600→        ctx.lineWidth = 3;
   601→   601→        ctx.setLineDash([9, 7]);
   602→   602→        ctx.beginPath();
   603→   603→        ctx.ellipse(worldX, GROUND_Y + 4, 46 * s, 11 * s, 0, 0, Math.PI * 2);
   604→   604→        ctx.stroke();
   605→   605→        ctx.restore();
   606→   606→      }
   607→   607→
   608→   608→      if (w) {
   609→   609→        this.drawSpeakerPill(design.name, worldX, s);
   610→   610→      }
   611→   611→    }
   612→   612→
   613→   613→    // walk-target flag (only while paused + an editable walk beat selected)
   614→   614→    if (!this.playing && this.editBeatId) {
   615→   615→      const w = wins.find((x) => x.beat.id === this.editBeatId);
   616→   616→      if (
   617→   617→        w &&
   618→   618→        (w.beat.action === 'walk' || w.beat.action === 'run') &&
   619→   619→        w.beat.moveTo != null
   620→   620→      ) {
   621→   621→        const actor = this.scene.characters.find((c) => c.uid === w.beat.actorUid);
   622→   622→        const fromX = actor
   623→   623→          ? this.charXAt(w.beat.actorUid, w.start, actor)
   624→   624→          : STAGE_W / 2;
   625→   625→        this.drawWalkFlag(fromX, w.beat.moveTo);
   626→   626→      }
   627→   627→    }
   628→   628→
   629→   629→    // subtitle: the most recently started active beat with dialogue
   630→   630→    let sub: BeatWindow | null = null;
   631→   631→    for (const w of active) {
   632→   632→      if (w.beat.text.trim() && (!sub || w.start > sub.start)) sub = w;
   633→   633→    }
   634→   634→    if (sub) {
   635→   635→      this.drawSubtitle(sub.beat.text);
   636→   636→    }
   637→   637→
   638→   638→    // watermark
   639→   639→    this.drawWatermark();
   640→   640→  }

   642→   642→  private drawWalkFlag(fromX: number, moveToPct: number) {

   716→   716→  private drawSubtitle(text: string) {
   717→   717→    const ctx = this.ctx;
   718→   718→    let fontSize = 30;
   719→   719→    ctx.font = `600 ${fontSize}px "Hind Siliguri", sans-serif`;
   720→   720→    const maxW = 980;
   721→   721→    let lines = wrapText(ctx, text, maxW);
   722→   722→    if (lines.length > 2) {
   723→   723→      fontSize = 25;
   724→   724→      ctx.font = `600 ${fontSize}px "Hind Siliguri", sans-serif`;
   725→   725→      lines = wrapText(ctx, text, maxW);
   726→   726→    }
   727→   727→    const lh = fontSize * 1.45;
   728→   728→    const boxH = lines.length * lh + 22;
   729→   729→    const boxW = Math.min(
   730→   730→      maxW + 56,
   731→   731→      Math.max(...lines.map((l) => ctx.measureText(l).width)) + 56,
   732→   732→    );
   733→   733→    const x0 = (STAGE_W - boxW) / 2;
   734→   734→    const y0 = STAGE_H - boxH - 14;
   735→   735→
   736→   736→    ctx.fillStyle = 'rgba(255, 251, 235, 0.95)';
   737→   737→    ctx.strokeStyle = '#f59e0b';
   738→   738→    ctx.lineWidth = 3;
   739→   739→    ctx.beginPath();
   740→   740→    ctx.roundRect(x0, y0, boxW, boxH, 14);
   741→   741→    ctx.fill();
   742→   742→    ctx.stroke();
   743→   743→
   744→   744→    ctx.fillStyle = '#1c1917';
   745→   745→    ctx.textAlign = 'center';
   746→   746→    ctx.textBaseline = 'middle';
   747→   747→    lines.forEach((line, i) => {
   748→   748→      ctx.fillText(line, STAGE_W / 2, y0 + 14 + lh / 2 + i * lh);
   749→   749→    });
   750→   750→  }
